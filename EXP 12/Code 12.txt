
#include <Arduino.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/semphr.h"

SemaphoreHandle_t BinarySemaphore;

void BinaryProducerTask(void *pvParameters)
{
  int eventCount = 1;

  while (1)
  {
    Serial.printf("[BINARY] Producer generated event %d\n",
                  eventCount);

    if (xSemaphoreGive(BinarySemaphore) == pdTRUE)
    {
      Serial.println("[BINARY] Event signal sent");
    }

    eventCount++;

    vTaskDelay(pdMS_TO_TICKS(1000));
  }
}

void BinaryConsumerTask(void *pvParameters)
{
  while (1)
  {
    if (xSemaphoreTake(BinarySemaphore, portMAX_DELAY) == pdTRUE)
    {
      Serial.println("[BINARY] Consumer received signal");
    }
  }
}

void setup()
{
  Serial.begin(115200);
  delay(1000);

  Serial.println("============================");
  Serial.println("[BINARY] SEMAPHORE EVENT DEMO");
  Serial.println("============================");

  BinarySemaphore = xSemaphoreCreateBinary();

  if (BinarySemaphore != NULL)
  {
    xTaskCreate(BinaryProducerTask, "Producer",
                2048, NULL, 2, NULL);

    xTaskCreate(BinaryConsumerTask, "Consumer",
                2048, NULL, 1, NULL);
  }
}

void loop()
{
  delay(1000);
}
