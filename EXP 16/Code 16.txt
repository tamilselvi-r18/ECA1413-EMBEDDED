
#include <Arduino.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/semphr.h"

SemaphoreHandle_t mutex;

void LowTask(void *pvParameters)
{
  Serial.println("[LOW] Task started. Priority = 1");

  xSemaphoreTake(mutex, portMAX_DELAY);

  Serial.println("[LOW] Shared resource LOCKED");

  vTaskDelay(pdMS_TO_TICKS(2500));

  Serial.print("[LOW] Current Priority = ");
  Serial.println(uxTaskPriorityGet(NULL));

  Serial.println("[LOW] Resource released");
  xSemaphoreGive(mutex);

  Serial.print("[LOW] Priority after release = ");
  Serial.println(uxTaskPriorityGet(NULL));

  vTaskDelete(NULL);
}

void MediumTask(void *pvParameters)
{
  while (1)
  {
    Serial.println("[MEDIUM] Medium priority task running");
    vTaskDelay(pdMS_TO_TICKS(400));
  }
}

void HighTask(void *pvParameters)
{
  vTaskDelay(pdMS_TO_TICKS(700));

  Serial.println("[HIGH] Requesting shared resource");

  xSemaphoreTake(mutex, portMAX_DELAY);

  Serial.println("[HIGH] Resource LOCKED");
  Serial.println("[HIGH] Processing complete");

  xSemaphoreGive(mutex);

  Serial.println("[HIGH] Resource released");

  vTaskDelete(NULL);
}

void setup()
{
  Serial.begin(115200);
  delay(1000);

  Serial.println("==============================");
  Serial.println("PRIORITY INHERITANCE TEST");
  Serial.println("==============================");

  mutex = xSemaphoreCreateMutex();

  if (mutex != NULL)
  {
    Serial.println("Mutex created successfully");

    xTaskCreatePinnedToCore(
      LowTask, "LowTask", 2048, NULL, 1, NULL, 1
    );

    delay(200);

    xTaskCreatePinnedToCore(
      MediumTask, "MediumTask", 2048, NULL, 2, NULL, 1
    );

    delay(200);

    xTaskCreatePinnedToCore(
      HighTask, "HighTask", 2048, NULL, 3, NULL, 1
    );
  }
}

void loop()
{
  delay(1000);
}
